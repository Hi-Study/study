// ShareDetail — Slack 메시지형 공유 글 상세. 공유자 메시지 + 링크 언퍼를(썸네일) + 리액션 + 댓글 메시지 행.
function ShareDetail({ study, shareId, shares, profileRole }) {
  const all = shares || study.shares;
  const s = all.find((x) => x.id === shareId) || all[0];
  const isText = !!s.body;
  const myRole = () => (profileRole || "").trim() || "멤버";

  const hue = (n) => { let h = 0; for (const ch of n) h = (h * 31 + ch.charCodeAt(0)) % 360; return h; };
  const Avatar = ({ name, size = 36 }) => (
    <div style={{ width: size, height: size, borderRadius: 8, background: `hsl(${hue(name)} 42% 42%)`, color: "#fff", display: "flex", alignItems: "center", justifyContent: "center", fontWeight: 700, fontSize: size * 0.42, flexShrink: 0, fontFamily: "var(--font-display)" }}>{name[0]}</div>
  );

  const EMOJI_UNUSED = null;
  const [postLiked, setPostLiked] = React.useState(false);
  const [comments, setComments] = React.useState([
    { id: 1, author: "한독자", role: "Junior PM", time: "1시간 전", text: "공유 감사합니다. 바로 읽어볼게요!", likes: 3, liked: false },
    { id: 2, author: "이지표", role: "Data PM", time: "40분 전", text: "저도 인상 깊게 읽었어요. 특히 마지막 문단.", likes: 7, liked: false },
  ]);
  const [draft, setDraft] = React.useState("");
  const [replyTo, setReplyTo] = React.useState(null);
  const [summary, setSummary] = React.useState(null);
  const [sumLoading, setSumLoading] = React.useState(false);
  const summarize = async () => {
    setSumLoading(true);
    try {
      const t = await window.claude.complete({ max_tokens: 500, messages: [{ role: "user", content: `\uc544\ub798 \uae00\uc744 \ud55c\uad6d\uc5b4\ub85c \uc694\uc57d\ud574\uc918. \ud615\uc2dd: \ud575\uc2ec 2~3\ubb38\uc7a5 + \ubd88\ub9bf 3\uac1c. \uc6d0\ubb38 \uc804\ubb38\uc740 \uc81c\uacf5\ub418\uc9c0 \uc54a\uc558\uc73c\ub2c8 \uc81c\ubaa9\u00b7\ucd9c\ucc98\u00b7\uba54\ubaa8\ub97c \uadfc\uac70\ub85c \ud575\uc2ec\uc744 \ucd94\uc815\ud558\ub418 \uacfc\uc7a5\ud558\uc9c0 \ub9d0\uace0, \ub9c8\uc9c0\ub9c9 \uc904\uc5d0 "\u203b \uc81c\ubaa9\u00b7\uba54\ubaa8 \uae30\ubc18 \ucd94\uc815 \uc694\uc57d\uc785\ub2c8\ub2e4"\ub77c\uace0 \uc801\uc5b4\uc918.\n\n\uc81c\ubaa9: ${s.title}\n\ucd9c\ucc98: ${s.source || "\uc9c1\uc811 \uc791\uc131"}\n\uacf5\uc720\uc790 \uba54\ubaa8: ${s.note || "(\uc5c6\uc74c)"}` }] });
      setSummary((t || "").trim());
    } catch (e) { setSummary("\uc694\uc57d\uc744 \ubd88\ub7ec\uc624\uc9c0 \ubabb\ud588\uc5b4\uc694. \uc7a0\uc2dc \ud6c4 \ub2e4\uc2dc \uc2dc\ub3c4\ud574\uc8fc\uc138\uc694."); }
    setSumLoading(false);
  };

  const toggleComment = (cid) => setComments(comments.map((c) => c.id === cid ? { ...c, liked: !c.liked, likes: c.likes + (c.liked ? -1 : 1) } : c));
  const add = () => { if (!draft.trim()) return; setComments([...comments, { id: Date.now(), author: "나", role: myRole(), time: "방금", text: draft.trim(), quote: replyTo ? { author: replyTo.author, text: replyTo.text } : null, likes: 0, liked: false }]); setDraft(""); setReplyTo(null); };

  const Thumb = ({ on }) => <svg width="14" height="14" viewBox="0 0 24 24" fill={on ? "currentColor" : "none"} stroke="currentColor" strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round"><path d="M7 10v12"/><path d="M15 5.88 14 10h5.83a2 2 0 0 1 1.92 2.56l-2.33 8A2 2 0 0 1 17.5 22H4a2 2 0 0 1-2-2v-8a2 2 0 0 1 2-2h2.76a2 2 0 0 0 1.79-1.11L12 2a3.13 3.13 0 0 1 3 3.88Z"/></svg>;
  const likeBtn = (on) => ({ display: "inline-flex", alignItems: "center", gap: 5, border: "none", background: "transparent", color: on ? "var(--color-primary)" : "var(--color-ink-muted-48)", fontSize: 12.5, fontWeight: 600, cursor: "pointer", padding: "0 4px" });

  return (
    <div style={{ minHeight: "100%", background: "var(--surface-card)", display: "flex", flexDirection: "column" }}>
      {/* 채널 헤더 */}
      <div style={{ padding: "12px 16px", borderBottom: "1px solid var(--color-hairline)" }}>
        <div style={{ fontSize: 11.5, color: "var(--color-ink-muted-48)", fontWeight: 700, textTransform: "uppercase", letterSpacing: ".6px" }}>공유 글{isText ? " · 직접 작성" : ""}</div>
        <h1 style={{ fontFamily: "var(--font-display)", fontSize: 18, fontWeight: 700, lineHeight: 1.3, letterSpacing: "-.3px", color: "var(--color-ink)", margin: "4px 0 0", textWrap: "balance" }}>{s.title}</h1>
      </div>

      <div style={{ flex: 1, padding: "10px 6px" }}>
        {/* 공유자 메시지 */}
        <div className="slk-msg" style={{ display: "flex", gap: 8, padding: "8px 10px", borderRadius: 8 }}>
          <Avatar name={s.sharedBy} />
          <div style={{ flex: 1, minWidth: 0 }}>
            <div style={{ display: "flex", alignItems: "baseline", gap: 7, flexWrap: "wrap" }}>
              <span style={{ fontFamily: "var(--font-display)", fontWeight: 700, fontSize: 15, color: "var(--color-ink)" }}>{s.sharedBy}</span>
              <span style={{ fontSize: 12, color: "var(--color-ink-muted-48)" }}>{s.role} · 글을 공유했어요</span>
            </div>
            {s.note && s.note.split("\n").filter(Boolean).map((p, i) => (
              <p key={i} style={{ fontSize: 15, lineHeight: 1.5, color: "var(--color-ink)", margin: i ? "4px 0 0" : "3px 0 0", textWrap: "pretty" }}>{p}</p>
            ))}

            {/* 링크 카드 (주소 비노출) */}
            {!isText && (
              <div style={{ margin: "10px 0 2px", paddingLeft: 12, borderLeft: "4px solid var(--color-primary)" }}>
                <div style={{ fontSize: 15, fontWeight: 700, color: "var(--color-ink)", lineHeight: 1.35, margin: "0 0 6px" }}>{s.title}</div>
                <div style={{ display: "flex", gap: 10, alignItems: "flex-start" }}>
                  <p style={{ flex: 1, fontSize: 13.5, color: "var(--color-ink-muted-80)", margin: 0, lineHeight: 1.5 }}>원문 전체 내용은 아래 버튼으로 확인하세요.</p>
                  <div style={{ width: 76, height: 56, borderRadius: 8, background: "var(--color-canvas-parchment)", border: "1px solid var(--color-hairline)", flexShrink: 0, display: "flex", alignItems: "center", justifyContent: "center", fontSize: 20 }}>🖼️</div>
                </div>
                <a href={s.url} target="_blank" rel="noopener noreferrer" style={{ display: "inline-flex", alignItems: "center", gap: 5, marginTop: 10, padding: "7px 14px", borderRadius: "var(--radius-pill)", background: "var(--action)", color: "#fff", fontSize: 12.5, fontWeight: 700, textDecoration: "none" }}>원문 보기 ↗</a>
              </div>
            )}

            {/* AI 요약 */}
            {!isText && (
              <div style={{ marginTop: 10, border: "1px solid var(--color-hairline)", borderRadius: 10, overflow: "hidden" }}>
                <div style={{ display: "flex", alignItems: "center", gap: 7, padding: "9px 12px", background: "var(--tint-lavender)" }}>
                  <span style={{ fontSize: 14 }}>✨</span>
                  <span style={{ fontSize: 13, fontWeight: 700, color: "var(--color-primary)", flex: 1 }}>AI 요약</span>
                  {!summary && <button onClick={summarize} disabled={sumLoading} style={{ border: "none", borderRadius: "var(--radius-pill)", background: "var(--color-primary)", color: "#fff", fontSize: 12, fontWeight: 700, padding: "5px 12px", cursor: sumLoading ? "default" : "pointer", opacity: sumLoading ? 0.6 : 1 }}>{sumLoading ? "요약 중…" : "요약하기"}</button>}
                </div>
                {summary && <div style={{ padding: "11px 13px" }}>{summary.split("\n").filter(Boolean).map((l, i) => <p key={i} style={{ fontSize: 13.5, lineHeight: 1.55, color: l.startsWith("※") ? "var(--color-ink-muted-48)" : "var(--color-ink)", margin: i ? "6px 0 0" : 0, textWrap: "pretty" }}>{l}</p>)}</div>}
              </div>
            )}

            {/* 직접 작성 본문 */}
            {isText && (
              <div style={{ margin: "8px 0 2px", padding: "12px 14px", border: "1px solid var(--color-hairline)", borderRadius: 10, background: "var(--surface-page)" }}>
                {s.body.map((p, i) => (
                  <p key={i} style={{ fontSize: 14.5, lineHeight: 1.6, color: "var(--color-ink)", margin: i ? "10px 0 0" : 0, textWrap: "pretty" }}>{p}</p>
                ))}
              </div>
            )}

            <div style={{ marginTop: 8 }}>
              <button onClick={() => setPostLiked(!postLiked)} style={likeBtn(postLiked)}><Thumb on={postLiked} /> 좋아요 {s.likes + (postLiked ? 1 : 0)}</button>
            </div>
          </div>
        </div>

        {/* 댓글 divider */}
        <div style={{ display: "flex", alignItems: "center", gap: 10, padding: "12px 12px 4px" }}>
          <span style={{ fontSize: 12.5, fontWeight: 700, color: "var(--color-ink-muted-48)" }}>답글 {comments.length}개</span>
          <span style={{ flex: 1, height: 1, background: "var(--color-hairline)" }} />
        </div>

        {/* 댓글 메시지 행 */}
        {comments.map((c) => (
          <div key={c.id} className="slk-msg" style={{ display: "flex", gap: 8, padding: "8px 10px", borderRadius: 8 }}>
            <Avatar name={c.author} />
            <div style={{ flex: 1, minWidth: 0 }}>
              <div style={{ display: "flex", alignItems: "baseline", gap: 7, flexWrap: "wrap" }}>
                <span style={{ fontFamily: "var(--font-display)", fontWeight: 700, fontSize: 15, color: "var(--color-ink)" }}>{c.author}</span>
                <span style={{ fontSize: 12, color: "var(--color-ink-muted-48)" }}>{c.role} · {c.time}</span>
              </div>
              {c.quote && (
                <div style={{ margin: "6px 0", paddingLeft: 9, borderLeft: "3px solid var(--color-hairline)" }}>
                  <span style={{ fontSize: 12.5, fontWeight: 700, color: "var(--color-ink-muted-48)" }}>{c.quote.author}</span>
                  <div style={{ fontSize: 13, color: "var(--color-ink-muted-48)", lineHeight: 1.45, overflow: "hidden", textOverflow: "ellipsis", display: "-webkit-box", WebkitLineClamp: 2, WebkitBoxOrient: "vertical" }}>{c.quote.text}</div>
                </div>
              )}
              <p style={{ fontSize: 15, lineHeight: 1.46, color: "var(--color-ink)", margin: "2px 0 0", textWrap: "pretty" }}>{c.text}</p>
              <div style={{ display: "flex", alignItems: "center", gap: 14, marginTop: 8 }}>
                <button onClick={() => toggleComment(c.id)} style={likeBtn(c.liked)}><Thumb on={c.liked} /> 좋아요 {c.likes}</button>
                <button onClick={() => setReplyTo(c)} style={{ display: "inline-flex", alignItems: "center", gap: 4, border: "none", background: "transparent", color: "var(--color-ink-muted-48)", fontSize: 12.5, fontWeight: 600, cursor: "pointer", padding: "0 4px" }}>
                  <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round"><path d="M9 17l-5-5 5-5"/><path d="M4 12h11a4 4 0 0 1 4 4v2"/></svg>답글
                </button>
              </div>
            </div>
          </div>
        ))}
      </div>

      {/* Slack 입력창 */}
      <div style={{ position: "sticky", bottom: 0, background: "var(--surface-card)", padding: "8px 12px 14px", borderTop: "1px solid var(--color-hairline)" }}>
        {replyTo && (
          <div style={{ display: "flex", alignItems: "center", gap: 8, background: "var(--tint-lavender)", borderRadius: 8, padding: "6px 10px", marginBottom: 6 }}>
            <span style={{ flex: 1, minWidth: 0, fontSize: 12.5, color: "var(--color-ink-muted-80)", overflow: "hidden", textOverflow: "ellipsis", whiteSpace: "nowrap" }}><b style={{ color: "var(--color-primary)" }}>{replyTo.author}</b>님에게 답글 · {replyTo.text}</span>
            <button onClick={() => setReplyTo(null)} aria-label="취소" style={{ border: "none", background: "transparent", color: "var(--color-ink-muted-48)", cursor: "pointer", fontSize: 16, lineHeight: 1 }}>✕</button>
          </div>
        )}
        <div style={{ display: "flex", alignItems: "flex-end", gap: 8, border: "1px solid var(--color-hairline)", borderRadius: 10, padding: "6px 8px 6px 12px", background: "var(--surface-card)" }}>
          <textarea value={draft} onChange={(e) => setDraft(e.target.value)} placeholder="댓글 보내기"
            style={{ flex: 1, minHeight: 24, maxHeight: 110, border: "none", outline: "none", resize: "none", fontFamily: "var(--font-text)", fontSize: 15, lineHeight: 1.5, color: "var(--color-ink)", background: "transparent", padding: "6px 0" }} />
          <button onClick={add} disabled={!draft.trim()} aria-label="보내기" style={{ width: 30, height: 30, borderRadius: 7, border: "none", cursor: draft.trim() ? "pointer" : "default", background: draft.trim() ? "var(--color-primary)" : "var(--color-canvas-parchment)", color: draft.trim() ? "#fff" : "var(--color-ink-muted-48)", display: "inline-flex", alignItems: "center", justifyContent: "center", flexShrink: 0 }}>
            <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><line x1="22" y1="2" x2="11" y2="13"/><polygon points="22 2 15 22 11 13 2 9 22 2"/></svg>
          </button>
        </div>
      </div>
    </div>
  );
}
window.ShareDetail = ShareDetail;
