// ProfileEdit — My > 프로필 설정. 표시 이름 + 직급 + 아이콘(첫 글자) 미리보기.
function ProfileEdit({ profileName, onChangeName, profileRole, onChangeRole }) {
  const shown = (profileName || "").trim() || "게스트";
  const field = { width: "100%", boxSizing: "border-box", padding: "13px 14px", border: "1px solid var(--color-hairline)", borderRadius: "var(--radius-md)", fontFamily: "var(--font-text)", fontSize: 16, color: "var(--color-ink)", outline: "none", background: "var(--surface-page)" };
  return (
    <div style={{ padding: "8px 16px 24px" }}>
      <div style={{ display: "flex", flexDirection: "column", alignItems: "center", padding: "12px 0 24px" }}>
        <span style={{ width: 84, height: 84, borderRadius: "50%", background: "var(--color-primary)", color: "#fff", display: "inline-flex", alignItems: "center", justifyContent: "center", fontFamily: "var(--font-display)", fontSize: 34, fontWeight: 600 }}>{shown[0]}</span>
        <div style={{ fontFamily: "var(--font-display)", fontSize: 18, fontWeight: 600, color: "var(--color-ink)", marginTop: 12 }}>{shown}</div>
        {profileRole ? <div style={{ fontSize: 13, color: "var(--color-ink-muted-48)", marginTop: 2 }}>{profileRole}</div> : null}
      </div>

      <div style={{ background: "var(--surface-card)", border: "1px solid var(--color-hairline)", borderRadius: "var(--radius-lg)", padding: 16 }}>
        <div style={{ fontSize: 12.5, color: "var(--color-ink-muted-48)", marginBottom: 8 }}>표시 이름</div>
        <input value={profileName} onChange={(e) => onChangeName(e.target.value)} placeholder="게스트" style={field} />
        <div style={{ fontSize: 12.5, color: "var(--color-ink-muted-48)", margin: "18px 0 8px" }}>직급 · 역할</div>
        <input value={profileRole} onChange={(e) => onChangeRole(e.target.value)} placeholder="예: PM, 기획자, 디자이너" style={field} />
        <div style={{ fontSize: 12, color: "var(--color-ink-muted-48)", marginTop: 10 }}>이름을 비우면 “게스트”로 표시되고, 아이콘은 이름 첫 글자를 사용해요. 직급은 내가 올린 글·댓글에 함께 표시됩니다.</div>
      </div>
    </div>
  );
}
window.ProfileEdit = ProfileEdit;
